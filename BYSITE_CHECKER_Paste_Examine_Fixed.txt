(function(){
'use strict';
const wafer=document.getElementById('wafer');
const product=document.getElementById('product');
const editor=document.querySelector('.editor');
const header=document.getElementById('wafer-heading');
const status=document.getElementById('status');
const retestAlert=document.getElementById('retest-alert');
const retestDetail=document.getElementById('retest-detail');
const upper=[...document.querySelectorAll('.editor [data-row]')];
const sheet=document.getElementById('input-grid');
const body=document.getElementById('input-body');
const head=document.getElementById('input-head');
const scroll=document.getElementById('sheet-scroll');
const rowsSaved={};
let selectedBefore='';
let rowCount=32, colCount=16;
let pasteShape=null; // Exact size of the single Excel rectangular paste used for product verification.
const activeSiteCount=()=>product.value==='proxima'?12:16;
let matrix=[];
function letter(n){let out='';for(n++;n;n=Math.floor((n-1)/26)){out=String.fromCharCode(65+(n-1)%26)+out}return out}
function ensure(r,c){
  let changed=false;
  while(rowCount<=r){rowCount+=16;changed=true}
  while(colCount<=c){colCount+=8;changed=true}
  while(matrix.length<rowCount)matrix.push([]);
  if(changed)build();
}
function build(){
  head.replaceChildren();
  const heading=document.createElement('tr');
  let corner=document.createElement('th');corner.className='row-num';corner.textContent='#';heading.append(corner);
  for(let c=0;c<colCount;c++){const th=document.createElement('th');th.textContent=letter(c);heading.append(th)}
  head.append(heading);
  const frag=document.createDocumentFragment();
  for(let r=0;r<rowCount;r++){
    const tr=document.createElement('tr');let num=document.createElement('th');num.className='row-num';num.textContent=r+1;tr.append(num);
    for(let c=0;c<colCount;c++){
      const td=document.createElement('td');td.contentEditable='true';td.spellcheck=false;
      td.dataset.r=r;td.dataset.c=c;
      td.textContent=(matrix[r]&&matrix[r][c])||'';
      tr.append(td);
    }
    frag.append(tr);
  }
  body.replaceChildren(frag);
}
function store(){if(!wafer.value)return;rowsSaved[wafer.value]={matrix:matrix.map(row=>row.slice()),upper:upper.map(c=>c.textContent),pasteShape:pasteShape?{...pasteShape}:null}}
function load(){
  const v=rowsSaved[wafer.value];
  matrix=v?v.matrix.map(row=>row.slice()):[];
  pasteShape=v&&v.pasteShape?{...v.pasteShape}:null;
  hideRetest();
  status.classList.remove('validation-error');
  const savedCols=matrix.reduce((a,row)=>Math.max(a,row.length),0);
  rowCount=Math.max(32,matrix.length);colCount=Math.max(16,savedCols);
  while(matrix.length<rowCount)matrix.push([]);
  build();
  upper.forEach((cell,i)=>{cell.textContent=v?v.upper[i]||'':'';cell.classList.remove('bad','good')});
  status.textContent='';
}
wafer.addEventListener('change',function(){
  const next=wafer.value;
  const previous=selectedBefore;
  wafer.value=previous;store();selectedBefore=next;
  wafer.value=next;
  // When users paste before choosing their first wafer, keep those pasted cells.
  if(!previous || (Object.keys(rowsSaved).length===0 && matrix.some(row=>row.some(v=>String(v??'').trim()!=='')))){
    store();
    status.textContent='Wafer selected. Tap Examine to update the upper table.';
  } else {
    load();
  }
});
sheet.addEventListener('input',e=>{
 const cell=e.target.closest('td[data-r]');if(!cell)return;
 const r=Number(cell.dataset.r),c=Number(cell.dataset.c);
 matrix[r][c]=cell.textContent.replace(/\r?\n/g,' ').trim();
 // Editing inside the original pasted block preserves its known column structure.
 // Editing outside the block makes its product identity uncertain.
 if(pasteShape && (r<pasteShape.startR || c<pasteShape.startC || c>=pasteShape.startC+pasteShape.width))pasteShape=null;
 store();
});
function parseExcel(text){
  // Retain trailing tab-delimited blank cells; remove only final line separators.
  const cleaned=String(text).replace(/\r\n?/g,'\n').replace(/\n+$/,'');
  if(!cleaned.trim())return null;
  const lines=cleaned.split('\n').map(line=>line.split('\t'));
  const width=Math.max(...lines.map(row=>row.length));
  return {lines,width,rectangular:lines.every(row=>row.length===width)};
}
function pasteGrid(text,startR,startC,replaceAll=false){
  const parsed=parseExcel(text);
  status.classList.remove('validation-error');
  if(!parsed){status.textContent='Nothing to load. Copy the Excel data, then press Ctrl+V inside the paste box.';return false;}
  if(!parsed.rectangular){
    status.textContent='PASTE ERROR: Excel rows have different column counts. Copy one complete rectangular Excel range, including the final columns.';
    status.classList.add('validation-error');return false;
  }
  if(replaceAll){
    // A new complete export replaces previous data and results, but preserves product and wafer selection.
    matrix=[];rowCount=32;colCount=16;pasteShape=null;clearResults();
    Object.keys(rowsSaved).forEach(k=>delete rowsSaved[k]);
  }
  // Product identity can be verified only for a full rectangular paste from cell A1.
  pasteShape=(startR===0 && startC===0 && (replaceAll || !pasteShape))

function numericValue(value){
  const raw=String(value??'').trim().replace(/,/g,'');
  if(raw==='')return null;
  const result=Number(raw);
  return Number.isFinite(result)?result:null;
}
function analyze(){
  let failures=0;
  const isProxima=product.value==='proxima';
  const excluded=isProxima?PROXIMA_EXCLUDED:CHANNING_EXCLUDED;
  const violationDetails=[];
  for(let site=0;site<activeSiteCount();site++){
    const qtyColumn=5+site*6;
    let maximum=null;
    let monitoredViolation=false;
    const matchingSbins=[];
    const violatingSbins=[];
    for(const row of matrix){
      const sbin=String(row[0]??'').trim();
      const binNumber=numericValue(sbin);
      if(excluded.has(binNumber))continue; // Ignore the entire row for this product.
      const qty=numericValue(row[qtyColumn]);
      if(qty===null)continue;
      // Display all eligible Sbins where the site's quantity is >=20.
      if(qty>=20 && sbin!=='')matchingSbins.push(sbin);
      if(maximum===null || qty>maximum)maximum=qty;
      if(isProxima && qty>=20 && PROXIMA_MONITORED.has(binNumber)){
        monitoredViolation=true;
        violatingSbins.push(sbin);
      }
    }
    const bysiteCell=document.querySelector(`[data-row="bysite"][data-site="${site}"]`);
    const maxCell=document.querySelector(`[data-row="max"][data-site="${site}"]`);
    const sbinCell=document.querySelector(`[data-row="sbin"][data-site="${site}"]`);
    const failed=maximum===null || (isProxima?monitoredViolation:maximum>=20);
    if(failed)failures++;
    bysiteCell.textContent=failed?'FAILED':'PASSED';
    bysiteCell.classList.toggle('bad',failed);
    bysiteCell.classList.toggle('good',!failed);
    maxCell.textContent=maximum===null?'':String(maximum);
    // Highlight Max Qty by the numeric threshold, independently from Proxima's monitored-bin failure rule.
    maxCell.classList.toggle('bad',maximum===null || maximum>=20);
    maxCell.classList.toggle('good',maximum!==null && maximum<20);
    sbinCell.textContent=maximum===null?'':matchingSbins.join(', ');
    sbinCell.title=sbinCell.textContent;
    if(isProxima && violatingSbins.length){
      violationDetails.push(`Site${site}: ${[...new Set(violatingSbins)].join(', ')}`);
    }
  }
  hideRetest();
  if(isProxima && violationDetails.length){
    retestAlert.hidden=false;
    retestDetail.textContent='Monitored Sbin violations (Qty ≥ 20): '+violationDetails.join(' | ');
  }
  return failures;
}

// Replace only the current clipboard contents with an empty string.
// Browsers may reject clipboard writing (especially when opened via file://).
// Try the modern API first, then a legacy copy command as a fallback.
async function blankClipboard(){
  if(navigator.clipboard && typeof navigator.clipboard.writeText==='function'){
    try { await navigator.clipboard.writeText(''); return true; } catch(error) { /* fallback below */ }
  }
  const previousFocus=document.activeElement;
  const blank=document.createElement('textarea');
  blank.value='';
  blank.setAttribute('aria-hidden','true');
  blank.setAttribute('readonly','');
  blank.style.cssText='position:fixed;left:-10000px;top:0;opacity:0;';
  document.body.appendChild(blank);
  blank.focus();
  blank.select();
  let copied=false;
  try { copied=!!document.execCommand('copy'); } catch(error) { copied=false; }
  blank.remove();
  if(previousFocus && typeof previousFocus.focus==='function'){
    try { previousFocus.focus({preventScroll:true}); } catch(error) { previousFocus.focus(); }
  }
  return copied;
}
